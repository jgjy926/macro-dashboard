# Zero-Cost Macro Forecasting Engine
#
# Deliberately small. Everything here is free and open source; there is no paid
# data vendor, no paid AI/ML API, and no hosted service in the dependency list.
#
# Note what is NOT here: no HTTP client library (urllib from the standard library
# is enough for a keyless CSV endpoint), no ORM (sqlite3 is in the standard
# library), no plotting library (the dashboard draws its own inline SVG), and no
# web framework (the front end is static files).

python-dotenv>=1.0     # optional .env loading; the engine works without it
pytest>=8.0            # test suite
